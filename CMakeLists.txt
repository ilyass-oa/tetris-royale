cmake_minimum_required(VERSION 3.22)
project(tetris-royale VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(TETRIS_BUILD_GUI "Build the SFML graphical client" ON)
option(TETRIS_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(TETRIS_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Werror=return-type)
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(nlohmann_json 3.10 QUIET CONFIG)
if(NOT TARGET nlohmann_json::nlohmann_json)
    add_library(tetris_json INTERFACE)
    target_include_directories(tetris_json INTERFACE "${PROJECT_SOURCE_DIR}/lib/nlohmann_json/single_include")
    add_library(nlohmann_json::nlohmann_json ALIAS tetris_json)
endif()
include(CTest)
add_subdirectory(src/shared)
add_subdirectory(src/server)
add_subdirectory(src/client/network)
if(TETRIS_BUILD_GUI)
    find_package(SFML 3 QUIET COMPONENTS Graphics Window System)
    if(NOT TARGET SFML::Graphics)
        include(FetchContent)
        set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
        set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
        set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 3.0.0 GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(SFML)
    endif()
    add_subdirectory(src/client/gui)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
